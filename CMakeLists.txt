cmake_minimum_required(VERSION 3.24)
project(iOSRemoteBridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(protocol-tests shared/protocol_tests.cpp)
add_test(NAME protocol COMMAND protocol-tests)
add_executable(frame-window-tests shared/frame_window_tests.cpp)
add_test(NAME frame-window COMMAND frame-window-tests)
add_executable(hid-report-tests shared/hid_report_tests.cpp)
add_test(NAME native-hid-reports COMMAND hid-report-tests)
add_executable(mouse-motion-tests shared/mouse_motion_tests.cpp)
add_test(NAME mouse-motion COMMAND mouse-motion-tests)
add_library(irb-sharpen STATIC client/windows/Sharpen.cpp)
add_executable(sharpen-tests shared/sharpen_tests.cpp)
target_link_libraries(sharpen-tests PRIVATE irb-sharpen)
add_test(NAME sharpening COMMAND sharpen-tests)
if(MSVC)
  target_compile_options(protocol-tests PRIVATE /W4 /WX)
  target_compile_options(frame-window-tests PRIVATE /W4 /WX)
  target_compile_options(hid-report-tests PRIVATE /W4 /WX)
  target_compile_options(mouse-motion-tests PRIVATE /W4 /WX)
  target_compile_options(irb-sharpen PRIVATE /W4 /WX)
  target_compile_options(sharpen-tests PRIVATE /W4 /WX)
else()
  target_compile_options(protocol-tests PRIVATE -Wall -Wextra -Werror)
  target_compile_options(frame-window-tests PRIVATE -Wall -Wextra -Werror)
  target_compile_options(hid-report-tests PRIVATE -Wall -Wextra -Werror)
  target_compile_options(mouse-motion-tests PRIVATE -Wall -Wextra -Werror)
  target_compile_options(irb-sharpen PRIVATE -Wall -Wextra -Werror)
  target_compile_options(sharpen-tests PRIVATE -Wall -Wextra -Werror)
endif()
option(IRB_BUILD_CLIENT "Build the native Windows receiver" OFF)
if(IRB_BUILD_CLIENT)
  add_subdirectory(client/windows)
endif()
