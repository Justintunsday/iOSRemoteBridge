if(NOT WIN32)
  message(FATAL_ERROR "The desktop client requires Windows")
endif()
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_path(AVCODEC_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
find_library(AVCODEC_LIBRARY avcodec REQUIRED)
find_library(AVUTIL_LIBRARY avutil REQUIRED)
find_library(SWSCALE_LIBRARY swscale REQUIRED)
add_library(irb-decoder STATIC Decoder.cpp)
target_include_directories(irb-decoder PUBLIC ${AVCODEC_INCLUDE_DIR})
target_link_libraries(irb-decoder PUBLIC ${AVCODEC_LIBRARY} ${AVUTIL_LIBRARY} ${SWSCALE_LIBRARY})
add_library(irb-renderer STATIC BufferedRenderer.cpp)
target_compile_definitions(irb-renderer PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(irb-renderer PUBLIC gdi32 user32)
add_executable(renderer-smoke RendererSmoke.cpp)
target_link_libraries(renderer-smoke PRIVATE irb-renderer)
add_test(NAME buffered-renderer COMMAND renderer-smoke)
add_executable(iOSRemoteBridge WIN32 Main.cpp Connection.cpp)
target_link_libraries(iOSRemoteBridge PRIVATE irb-decoder irb-renderer OpenSSL::SSL OpenSSL::Crypto nlohmann_json::nlohmann_json ws2_32 comdlg32 shell32)
target_compile_definitions(iOSRemoteBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(iOSRemoteBridge PRIVATE ../../shared)
add_executable(decoder-smoke DecoderSmoke.cpp)
target_link_libraries(decoder-smoke PRIVATE irb-decoder)
add_executable(network-smoke NetworkSmoke.cpp Connection.cpp)
target_link_libraries(network-smoke PRIVATE irb-decoder OpenSSL::SSL OpenSSL::Crypto nlohmann_json::nlohmann_json ws2_32)
target_include_directories(network-smoke PRIVATE ../../shared)
target_compile_definitions(network-smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(iOSRemoteBridge PRIVATE /W4 /EHsc /utf-8)
  target_compile_options(irb-decoder PRIVATE /W4 /EHsc /utf-8)
endif()
