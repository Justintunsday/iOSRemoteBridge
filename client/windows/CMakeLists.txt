if(NOT WIN32)
  message(FATAL_ERROR "The desktop client requires Windows")
endif()
enable_language(RC)
configure_file(shaders/Resources.rc.in ${CMAKE_CURRENT_BINARY_DIR}/FSRResources.rc @ONLY)
add_library(irb-presenter STATIC D3DRenderer.cpp ExperimentalFSR.cpp)
target_compile_definitions(irb-presenter PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(irb-presenter PUBLIC d3d11 d3d12 dxgi d3dcompiler user32)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_path(AVCODEC_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
find_library(AVCODEC_LIBRARY avcodec REQUIRED)
find_library(AVUTIL_LIBRARY avutil REQUIRED)
find_library(SWSCALE_LIBRARY swscale REQUIRED)
add_library(irb-decoder STATIC Decoder.cpp ColorConverter.cpp)
target_include_directories(irb-decoder PUBLIC ${AVCODEC_INCLUDE_DIR})
target_link_libraries(irb-decoder PUBLIC ${AVCODEC_LIBRARY} ${AVUTIL_LIBRARY} ${SWSCALE_LIBRARY})
add_library(irb-renderer STATIC BufferedRenderer.cpp)
target_compile_definitions(irb-renderer PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(irb-renderer PUBLIC irb-sharpen gdi32 user32)
add_executable(renderer-smoke RendererSmoke.cpp)
target_link_libraries(renderer-smoke PRIVATE irb-renderer)
add_test(NAME buffered-renderer COMMAND renderer-smoke)
add_executable(iOSRemoteBridge WIN32 Main.cpp Connection.cpp ${CMAKE_CURRENT_BINARY_DIR}/FSRResources.rc)
target_link_libraries(iOSRemoteBridge PRIVATE irb-decoder irb-renderer irb-presenter OpenSSL::SSL OpenSSL::Crypto nlohmann_json::nlohmann_json ws2_32 comdlg32 shell32 comctl32)
add_executable(irb-gpu-check GPUSmoke.cpp ${CMAKE_CURRENT_BINARY_DIR}/FSRResources.rc)
target_link_libraries(irb-gpu-check PRIVATE irb-presenter)
add_test(NAME gpu-presentation COMMAND irb-gpu-check --warp)
add_executable(irb-fsr-check ExperimentalFSRCheck.cpp)
target_link_libraries(irb-fsr-check PRIVATE irb-presenter)
add_test(NAME experimental-fsr-runtime COMMAND irb-fsr-check --probe)
target_compile_definitions(iOSRemoteBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_include_directories(iOSRemoteBridge PRIVATE ../../shared)
add_executable(decoder-smoke DecoderSmoke.cpp)
target_link_libraries(decoder-smoke PRIVATE irb-decoder)
add_executable(color-smoke ColorSmoke.cpp)
target_link_libraries(color-smoke PRIVATE irb-decoder)
add_test(NAME video-color COMMAND color-smoke)
add_executable(input-delivery-check InputDeliveryCheck.cpp)
target_link_libraries(input-delivery-check PRIVATE nlohmann_json::nlohmann_json ws2_32)
target_compile_definitions(input-delivery-check PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME input-delivery COMMAND input-delivery-check)
add_executable(irb-color-check ColorDecodeCheck.cpp)
target_link_libraries(irb-color-check PRIVATE irb-decoder)
add_executable(network-smoke NetworkSmoke.cpp Connection.cpp)
target_link_libraries(network-smoke PRIVATE irb-decoder OpenSSL::SSL OpenSSL::Crypto nlohmann_json::nlohmann_json ws2_32)
target_include_directories(network-smoke PRIVATE ../../shared)
target_compile_definitions(network-smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(iOSRemoteBridge PRIVATE /W4 /EHsc /utf-8)
  target_compile_options(irb-decoder PRIVATE /W4 /EHsc /utf-8)
  target_compile_options(irb-presenter PRIVATE /W4 /EHsc /utf-8)
endif()
